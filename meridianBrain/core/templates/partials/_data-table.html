{# Data Table Macro Renders an HTML table with a styled header row and data rows. Parameters: columns - Array of column objects: [{key: "id", header: "ID", width: "60px"}, ...] rows - Array of row objects: [{id: "A1", name: "..."}, ...] headerBg - Optional header background color (defaults to "#f8f9fa") Column objects may include: key - Property name to look up in each row header - Display text for the column header width - Optional CSS width (e.g. "60px", "100px") #} {% macro dataTable(columns, rows, headerBg) -%} {%- for col in columns %} {%- endfor %} {%- for row in rows %} {%- for col in columns %} {%- endfor %} {%- endfor %}
{{ col.header }}
{{ row[col.key] }}
{%- endmacro %}
